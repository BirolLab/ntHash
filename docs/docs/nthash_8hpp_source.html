928 bool is_care_block = padded_string[0] == '1';
929 for (unsigned pos = 0; pos < padded_string.length(); pos++) {
930 if (is_care_block && padded_string[pos] == '0') {
931 if (pos - i_start == 1) {
932 care_monos.push_back(i_start);
933 } else {
934 care_blocks.push_back({ i_start, pos });
935 }
936 i_start = pos;
937 is_care_block = false;
938 } else if (!is_care_block && padded_string[pos] == '1') {
939 if (pos - i_start == 1) {
940 ignore_monos.push_back(i_start);
941 } else {
942 ignore_blocks.push_back({ i_start, pos });
943 }
944 i_start = pos;
945 is_care_block = true;
946 }
947 }
948
949 const unsigned num_cares = care_blocks.size() * 2 + care_monos.size();
950 const unsigned num_ignores =
951 ignore_blocks.size() * 2 + ignore_monos.size() + 2;
952
953 if (num_ignores < num_cares) {
954 ignore_blocks.push_back(
955 { 0, static_cast<unsigned>(seed_string.length()) });
956 blocks.push_back(ignore_blocks);
957 monomers.push_back(ignore_monos);
958 } else {
959 blocks.push_back(care_blocks);
960 monomers.push_back(care_monos);
961 }
962 }

1002inline bool
1003ntmsm64_init(const char* kmer_seq,
1004 const std::vector<SpacedSeedBlocks>& seeds_blocks,
1005 const std::vector<SpacedSeedMonomers>& seeds_monomers,
1006 K_TYPE k,
1007 unsigned m,
1008 unsigned m2,
1009 const std::vector<std::array<HASH_TYPE, 4>>& fwd_shift_table,
1010 const std::vector<std::array<HASH_TYPE, 4>>& rev_shift_table,
1011 HASH_TYPE* fh_nomonos,
1012 HASH_TYPE* rh_nomonos,
1013 HASH_TYPE* fh_val,
1014 HASH_TYPE* rh_val,
1015 HASH_TYPE* h_val)
1016{
1017 for (unsigned i_seed = 0; i_seed < m; i_seed++) {
1018 HASH_TYPE fh_seed = 0;
1019 HASH_TYPE rh_seed = 0;
1020
1021 for (const auto& block : seeds_blocks[i_seed]) {
1022 for (unsigned pos = block[0]; pos < block[1]; pos++) {
1023 const auto c = static_cast<unsigned char>(kmer_seq[pos]);
1024 const auto idx = internal::CONVERT_TAB[c];
1025 fh_seed ^= fwd_shift_table[k - 1 - pos][idx];
1026 rh_seed ^= rev_shift_table[pos][idx];
1027 }
1028 }
1029
1030 fh_nomonos[i_seed] = fh_seed;
1031 rh_nomonos[i_seed] = rh_seed;
1032
1033 for (const auto pos : seeds_monomers[i_seed]) {
1034 const auto c = static_cast<unsigned char>(kmer_seq[pos]);
1035 const auto idx = internal::CONVERT_TAB[c];
1036 fh_seed ^= fwd_shift_table[k - 1 - pos][idx];
1037 rh_seed ^= rev_shift_table[pos][idx];
1038 }
1039
1040 fh_val[i_seed] = fh_seed;
1041 rh_val[i_seed] = rh_seed;
1042
1043 const auto base = i_seed * m2;
1044 h_val[base] = internal::canonical(fh_seed, rh_seed);
1045 for (unsigned i_hash = 1; i_hash < m2; i_hash++) {
1046 auto t = h_val[base] *
1047 (i_hash ^ static_cast<HASH_TYPE>(k) * internal::MULTISEED);
1048 t ^= t >> internal::MULTISHIFT;
1049 h_val[base + i_hash] = t;
1050 }
1051 }
1052
1053 return true;
